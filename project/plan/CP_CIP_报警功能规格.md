## 3. PLC报警处理

### 3.1 报警触发 (FC_TriggerAlarm)

```pascal
FUNCTION "FC_TriggerAlarm" : Void
VAR_INPUT
    AlarmID : Int;
    Level : "Enum_AlarmLevel";
    ZoneID : Int;
END_VAR
VAR
    i : Int;
    Slot : Int;
    AlreadyActive : Bool;
END_VAR

    // ========================================================================
    // 1. 检查该报警是否已经激活(去重)
    // ========================================================================
    AlreadyActive := FALSE;
    FOR i := 1 TO 50 DO
        IF "Global".AlarmActive[i].AlarmID = AlarmID
        AND "Global".AlarmActive[i].ZoneID = ZoneID
        AND "Global".AlarmActive[i].Status = 0 THEN  // 0=激活
            AlreadyActive := TRUE;
            EXIT;
        END_IF;
    END_FOR;

    // 如果报警已激活，不再重复触发
    IF AlreadyActive THEN
        RETURN;
    END_IF;

    // ========================================================================
    // 2. 查找空闲报警槽
    // ========================================================================
    Slot := 0;
    FOR i := 1 TO 50 DO
        IF "Global".AlarmActive[i].Status <> 0 THEN
            Slot := i;
            EXIT;
        END_IF;
    END_FOR;

    // 如果没有空闲槽，覆盖最早的报警
    IF Slot = 0 THEN
        Slot := 1;
    END_IF;

    // ========================================================================
    // 3. 填充报警信息
    // ========================================================================
    "Global".AlarmActive[Slot].AlarmID := AlarmID;
    "Global".AlarmActive[Slot].Level := Level;
    "Global".AlarmActive[Slot].ZoneID := ZoneID;
    "Global".AlarmActive[Slot].TriggerTime := "Global".System.SystemTime;
    "Global".AlarmActive[Slot].Status := 0;  // 激活状态

    // 根据报警ID填充报警代码和文本
    CASE AlarmID OF
        1:  "Global".AlarmActive[Slot].AlarmCode := 1;
            "Global".AlarmActive[Slot].AlarmText := '碱液罐液位低';
        2:  "Global".AlarmActive[Slot].AlarmCode := 2;
            "Global".AlarmActive[Slot].AlarmText := '酸液罐液位低';
        3:  "Global".AlarmActive[Slot].AlarmCode := 3;
            "Global".AlarmActive[Slot].AlarmText := '热水罐液位低';
        4:  "Global".AlarmActive[Slot].AlarmCode := 4;
            "Global".AlarmActive[Slot].AlarmText := '纯水罐液位低';
        5:  "Global".AlarmActive[Slot].AlarmCode := 5;
            "Global".AlarmActive[Slot].AlarmText := '清洗温度不达标';
        6:  "Global".AlarmActive[Slot].AlarmCode := 6;
            "Global".AlarmActive[Slot].AlarmText := '电导率超标';
        7:  "Global".AlarmActive[Slot].AlarmCode := 7;
            "Global".AlarmActive[Slot].AlarmText := '温度超限';
        8:  "Global".AlarmActive[Slot].AlarmCode := 8;
            "Global".AlarmActive[Slot].AlarmText := '管路压力异常';
        9:  "Global".AlarmActive[Slot].AlarmCode := 9;
            "Global".AlarmActive[Slot].AlarmText := '清洗泵过载';
        12..16:
            "Global".AlarmActive[Slot].AlarmCode := AlarmID;
            "Global".AlarmActive[Slot].AlarmText := CONCAT('区无流量', INT_TO_STRING(AlarmID - 11));
        17:
            "Global".AlarmActive[Slot].AlarmCode := 17;
            "Global".AlarmActive[Slot].AlarmText := '急停触发';
        18:
            "Global".AlarmActive[Slot].AlarmCode := 18;
            "Global".AlarmActive[Slot].AlarmText := '清洗序列非法';
    END_CASE;

    // ========================================================================
    // 4. 增加报警计数
    // ========================================================================
    "Global".AlarmCount := "Global".AlarmCount + 1;
    "Global".System.AlarmCount := "Global".AlarmCount;

END_FUNCTION
```

### 3.2 报警检测 (FC_CheckAlarms)

```pascal
FUNCTION "FC_CheckAlarms" : Void
VAR_TEMP
    i : Int;
END_VAR

    // ========================================================================
    // 罐区液位检测 (L2)
    // ========================================================================
    IF NOT "Global".TankAlkali.LevelLSL THEN
        "FC_TriggerAlarm"(1, "Enum_AlarmLevel".L2_INFO, 0);
    END_IF;
    IF NOT "Global".TankAcid.LevelLSL THEN
        "FC_TriggerAlarm"(2, "Enum_AlarmLevel".L2_INFO, 0);
    END_IF;
    IF NOT "Global".TankHotWater.LevelLSL THEN
        "FC_TriggerAlarm"(3, "Enum_AlarmLevel".L2_INFO, 0);
    END_IF;
    IF NOT "Global".TankPureWater.LevelLSL THEN
        "FC_TriggerAlarm"(4, "Enum_AlarmLevel".L2_INFO, 0);
    END_IF;

    // ========================================================================
    // 温度不达标检测 (L1) - 在FC_StepMonitor中单独处理
    // ========================================================================

    // ========================================================================
    // 泵过载检测 (L0)
    // ========================================================================
    FOR i := 1 TO 5 DO
        IF "Global".DI_PumpOverload[i] THEN
            "FC_TriggerAlarm"(9, "Enum_AlarmLevel".L0_CRITICAL, i);
            "DB_Zone"[i].PumpState := "Enum_PumpState".PUMP_FAULT;
        END_IF;
    END_FOR;

    // ========================================================================
    // 急停检测 (L0)
    // ========================================================================
    IF "Global".System.EStop THEN
        "FC_TriggerAlarm"(17, "Enum_AlarmLevel".L0_CRITICAL, 0);
    END_IF;

END_FUNCTION
```

### 3.3 报警处理 (FC_AlarmHandler)

```pascal
FUNCTION "FC300_AlarmHandler" : Void

    // ========================================================================
    // 1. 蜂鸣器控制
    // ========================================================================
    IF "Global".AlarmCount > 0 THEN
        // 检查是否有未确认的L0/L1报警
        IF "FC_HasHighPriorityAlarm"() THEN
            "Global".Output.Buzzer := TRUE;  // 连续响
        ELSE
            "Global".Output.Buzzer := FALSE;
        END_IF;
    ELSE
        "Global".Output.Buzzer := FALSE;
    END_IF;

    // ========================================================================
    // 2. L0报警自动处理 (急停相关)
    // ========================================================================
    IF "FC_IsAlarmActive"(17) THEN  // 急停报警
        "FC_EmergencyStop"();
    END_IF;

END_FUNCTION
```

## 8. 通讯接口

### 8.1 报警状态推送

```typescript
interface AlarmMessage {
  type: 'alarm';
  alarm_id: number;
  alarm_code: string;      // 'CP-001'
  level: string;            // 'L0_CRITICAL' | 'L1_WARNING' | 'L2_INFO'
  level_color: string;      // 颜色代码
  zone_id: number;
  zone_name: string;
  text: string;
  trigger_time: string;    // ISO8601
  ack_time?: string;
  ack_user?: string;
  status: string;           // 'ACTIVE' | 'ACKED' | 'CLEARED'
}

interface AlarmSummaryMessage {
  type: 'alarm_summary';
  total_count: number;
  by_level: {
    L0: number;
    L1: number;
```
